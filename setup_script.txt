#!/bin/bash

# Telegram Downloader Bot - Setup Script
# This script automates the installation process

set -e  # Exit on error

# Colors for output
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

# Functions
print_success() {
    echo -e "${GREEN}✓ $1${NC}"
}

print_error() {
    echo -e "${RED}✗ $1${NC}"
}

print_info() {
    echo -e "${YELLOW}ℹ $1${NC}"
}

# Check if running as root
if [ "$EUID" -eq 0 ]; then 
   print_error "Please do not run this script as root!"
   exit 1
fi

print_info "Starting Telegram Downloader Bot setup..."

# Check Python version
PYTHON_VERSION=$(python3 -c 'import sys; print(".".join(map(str, sys.version_info[:2])))')
REQUIRED_VERSION="3.8"

if [ "$(printf '%s\n' "$REQUIRED_VERSION" "$PYTHON_VERSION" | sort -V | head -n1)" != "$REQUIRED_VERSION" ]; then
    print_error "Python $REQUIRED_VERSION or higher is required. Found: Python $PYTHON_VERSION"
    exit 1
fi
print_success "Python version check passed: $PYTHON_VERSION"

# Install system dependencies
print_info "Installing system dependencies..."
sudo apt update
sudo apt install -y python3-pip python3-venv git libmagic1 libmagic-dev \
    libjpeg-dev zlib1g-dev libpng-dev libfreetype6-dev build-essential

print_success "System dependencies installed"

# Create virtual environment
print_info "Creating virtual environment..."
python3 -m venv venv
source venv/bin/activate

print_success "Virtual environment created"

# Upgrade pip
print_info "Upgrading pip..."
pip install --upgrade pip

# Install Python dependencies
print_info "Installing Python dependencies..."
pip install -r requirements.txt

# Remove Windows-specific package on Linux
pip uninstall -y python-magic-bin 2>/dev/null || true

print_success "Python dependencies installed"

# Create necessary directories
print_info "Creating directories..."
mkdir -p downloads uploads temp logs backups
chmod 755 downloads uploads temp logs backups

print_success "Directories created"

# Create .env file if it doesn't exist
if [ ! -f .env ]; then
    print_info "Creating .env file..."
    cp .env.example .env
    print_info "Please edit .env file with your bot token and admin ID"
else
    print_info ".env file already exists"
fi

# Create systemd service file
print_info "Creating systemd service file..."
SERVICE_FILE="/tmp/telegram-bot.service"
cat > $SERVICE_FILE << EOF
[Unit]
Description=Telegram Downloader Bot
After=network.target

[Service]
Type=simple
User=$USER
Group=$USER
WorkingDirectory=$PWD
Environment="PATH=$PWD/venv/bin"
EnvironmentFile=$PWD/.env
ExecStart=$PWD/venv/bin/python $PWD/telegram_bot.py
Restart=always
RestartSec=10
StandardOutput=append:$PWD/logs/bot.log
StandardError=append:$PWD/logs/error.log

[Install]
WantedBy=multi-user.target
EOF

print_info "Service file created at $SERVICE_FILE"
print_info "To install as a service, run:"
echo "  sudo cp $SERVICE_FILE /etc/systemd/system/"
echo "  sudo systemctl daemon-reload"
echo "  sudo systemctl enable telegram-bot"
echo "  sudo systemctl start telegram-bot"

# Create monitoring script
print_info "Creating monitoring script..."
cat > monitor.sh << 'EOF'
#!/bin/bash

# Read environment variables
source .env

SERVICE_NAME="telegram-bot.service"

# Check if service is running
if ! systemctl is-active --quiet $SERVICE_NAME; then
    echo "$(date): Bot is down! Attempting restart..."
    sudo systemctl restart $SERVICE_NAME
    
    # Send notification
    if [ ! -z "$BOT_TOKEN" ] && [ ! -z "$ADMIN_IDS" ]; then
        ADMIN_ID=$(echo $ADMIN_IDS | cut -d',' -f1)
        curl -s -X POST "https://api.telegram.org/bot$BOT_TOKEN/sendMessage" \
             -d "chat_id=$ADMIN_ID&text=⚠️ Bot was down and has been restarted!"
    fi
fi

# Log status
echo "$(date): Bot status checked - $(systemctl is-active $SERVICE_NAME)"
EOF

chmod +x monitor.sh
print_success "Monitoring script created"

# Create backup script
print_info "Creating backup script..."
cat > backup.sh << 'EOF'
#!/bin/bash

BACKUP_DIR="backups"
DATE=$(date +%Y%m%d_%H%M%S)

# Create backup directory
mkdir -p $BACKUP_DIR

# Backup database
if [ -f bot_database.db ]; then
    cp bot_database.db $BACKUP_DIR/db_backup_$DATE.db
    echo "Database backed up to $BACKUP_DIR/db_backup_$DATE.db"
fi

# Backup .env file
if [ -f .env ]; then
    cp .env $BACKUP_DIR/env_backup_$DATE
    echo "Environment file backed up"
fi

# Remove old backups (older than 30 days)
find $BACKUP_DIR -type f -mtime +30 -delete
echo "Old backups cleaned up"
EOF

chmod +x backup.sh
print_success "Backup script created"

# Summary
echo ""
print_success "Setup completed successfully!"
echo ""
echo "Next steps:"
echo "1. Edit .env file with your bot token and admin ID:"
echo "   nano .env"
echo ""
echo "2. Test the bot:"
echo "   source venv/bin/activate"
echo "   python telegram_bot.py"
echo ""
echo "3. (Optional) Install as systemd service:"
echo "   sudo cp /tmp/telegram-bot.service /etc/systemd/system/"
echo "   sudo systemctl daemon-reload"
echo "   sudo systemctl enable telegram-bot"
echo "   sudo systemctl start telegram-bot"
echo ""
echo "4. (Optional) Add monitoring to crontab:"
echo "   crontab -e"
echo "   Add: */5 * * * * $PWD/monitor.sh >> $PWD/logs/monitor.log 2>&1"
echo "   Add: 0 2 * * * $PWD/backup.sh >> $PWD/logs/backup.log 2>&1"
echo ""
print_info "For more information, see INSTALL.md"
